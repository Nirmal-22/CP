class Solution {\npublic:\n\tint numIslands(vector<vector<char>>& grid) {\n\t\tint m = grid.size();\n\t\tif(m==0) return 0;\n\t\tint n= grid[0].size(), count=0;\n\t\tfunction<void(int, int)> dfs = [&](int r, int c){\n\t\t\tif(r<0 || r>=m || c<0 || c>=n || grid[r][c] !='1')\n\t\t\t\treturn;\n\t\t\tgrid[r][c] = '0';\n\t\t\tdfs(r+1, c);\n\t\t\tdfs(r-1, c);\n\t\t\tdfs(r, c+1);\n\t\t\tdfs(r, c-1);\n\t\t};\n\t\tfor(int i=0;i<m;i++){\n\t\t\tfor(int j=0; j<n; j++){\n\t\t\t\tif(grid[i][j] == '1'){\n\t\t\t\t\t++count;\n\t\t\t\t\tdfs(i, j);\n\t\t\t\t}\n\t\t\t}\n\t\t}\n\t\treturn count;\n\t}\n};
