class Solution {\npublic:\n\tint numDecodings(string s) {\n\t\tint n= s.size();\n\t\tif(n==0 || s[0]=='0') return 0;\n\t\tvector<int> dp(n+1, 0);\n\t\tdp[0] = 1;\n\t\tdp[1] = 1;\n\t\tfor(int i=2;i<=n;i++){\n\t\t\tif(s[i-1]!='0')\n\t\t\t\tdp[i] += dp[i-1];\n\t\t\tint twoD = (s[i-2]-'0')*10 + (s[i-1]-'0');\n\t\t\tif(twoD>=10 && twoD<=26)\n\t\t\t\tdp[i]+=dp[i-2];\n\t\t}\n\t\treturn dp[n];\n\t}\n};
