class Solution {\npublic:\n\tint longestCommonSubsequence(string s1, string s2) {\n\t\tint m=s1.size(), n=s2.size();\n\t\tvector<vector<int>> dp(m+1, vector<int>(n+1,0));\n\t\tfor(int i=1;i<=m;i++){\n\t\t\tfor(int j=1;j<=n;j++){\n\t\t\t\tif(s1[i-1] == s2[j-1])\n\t\t\t\t\tdp[i][j] = dp[i-1][j-1] + 1;\n\t\t\t\telse\n\t\t\t\t\tdp[i][j] = max(dp[i-1][j], dp[i][j-1]);\n\t\t\t}\n\t\t}\n\t\treturn dp[m][n];\n\t}\n};
