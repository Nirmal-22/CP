class Solution {\npublic:\n\tbool isValid(string s) {\n\t\tstack<char> st;\n\t\tfor(char c:s){\n\t\t\tif(c=='(' || c=='[' || c=='{')\n\t\t\t\tst.push(c);\n\t\t\telse\n\t\t\t{\n\t\t\t\tif(st.empty()) return false;\n\t\t\t\tchar t = st.top(); st.pop();\n\t\t\t\tif((c==')' && t!='(') || (c==']' && t!='[') || (c=='}' && t!='{'))\n\t\t\t\t\treturn false;\n\t\t\t}\n\t\t}\n\t\treturn st.empty();\n\t}\n};
