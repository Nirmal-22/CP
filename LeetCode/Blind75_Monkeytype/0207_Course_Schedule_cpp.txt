class Solution {\npublic:\n\tbool canFinish(int n, vector<vector<int>>& g) {\n\t\tvector<vector<int>> adj(n);\n\t\tvector<int> id(n,0);\n\t\tfor(auto& p:g){\n\t\t\tint c = p[0], pre = p[1];\n\t\t\tadj[pre].push_back(c);\n\t\t\t++id[c];\n\t\t}\n\t\tqueue<int> q;\n\t\tfor(int i=0;i<n;i++){\n\t\t\tif(id[i] == 0) q.push(i);\n\t\t}\n\t\tint taken =0;\n\t\twhile(!q.empty()){\n\t\t\tint u=q.front(); q.pop();\n\t\t\t++taken;\n\t\t\tfor(int v: adj[u]){\n\t\t\t\tif(--id[v]==0){\n\t\t\t\t\tq.push(v);\n\t\t\t\t}\n\t\t\t}\n\t\t}\n\t\treturn (taken == n);\n\t}\n};
