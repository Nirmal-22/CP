class Solution {\npublic:\n\tint maxProfit(vector<int>& prices) {\n\t\tif (prices.empty())\n\t\treturn 0;\n\tint n = prices.size();\n\tstd::vector<int> dp(n, 0);\n\tint minPrice = prices[0];\n\tfor (int i = 1; i < n; ++i)\n\t{\n\t\tint currentProfit = prices[i] - minPrice;\n\t\tdp[i] = std::max(dp[i - 1], currentProfit);\n\t\tminPrice = std::min(minPrice, prices[i]);\n\t}\n\treturn dp[n - 1];\n\t}\n};
