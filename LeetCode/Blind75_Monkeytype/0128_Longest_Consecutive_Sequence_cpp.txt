class Solution {\npublic:\n\tint longestConsecutive(vector<int>& nums) {\n\t\tunordered_set<int> s(nums.begin(), nums.end());\n\t\tint longest = 0;\n\t\tfor(int num:s){\n\t\t\tif(!s.count(num-1)){\n\t\t\t\tint curr=num, streak=1;\n\t\t\t\twhile(s.count(curr+1)){\n\t\t\t\t\tcurr++;\n\t\t\t\t\tstreak++;\n\t\t\t\t}\n\t\t\t\tlongest = max(longest, streak);\n\t\t\t}\n\t\t}\n\t\treturn longest;\n\t}\n};
