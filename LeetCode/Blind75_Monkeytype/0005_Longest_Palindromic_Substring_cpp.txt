class Solution {\npublic:\n\tstring longestPalindrome(string s) {\n\t\tint n = s.size();\n\t\tif(n<2) return s;\n\t\tint start=0, maxlen=1;\n\t\tfor(int i=0;i<n;i++){\n\t\t\tint l=i,r=i;\n\t\t\twhile(l>=0 && r<n && s[l]==s[r])\n\t\t\t{\n\t\t\t\tif(r-l+1 > maxlen){\n\t\t\t\t\tstart=l;\n\t\t\t\t\tmaxlen=r-l+1;\n\t\t\t\t}\n\t\t\t\t--l,++r;\n\t\t\t}\n\t\t\tl=i,r=i+1;\n\t\t\twhile(l>=0 && r<n && s[l]==s[r]){\n\t\t\t\tif(r-l+1 > maxlen){\n\t\t\t\t\tstart=l;\n\t\t\t\t\tmaxlen = r-l+1;\n\t\t\t\t}\n\t\t\t\t--l,++r;\n\t\t\t}\n\t\t}\n\t\treturn s.substr(start, maxlen);\n\t}\n};
