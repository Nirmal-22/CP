class Solution {\npublic:\n\tListNode* removeNthFromEnd(ListNode* head, int n) {\n\t\tListNode dummy(0);\n\t\tdummy.next = head;\n\t\tListNode * fast=&dummy, *slow=&dummy;\n\t\tfor(int i=0;i<=n;i++) fast=fast->next;\n\t\twhile(fast) {\n\t\t\tfast = fast->next;\n\t\t\tslow = slow->next;\n\t\t}\n\t\tListNode* to_rem = slow->next;\n\t\tslow->next = to_rem->next;\n\t\tdelete to_rem;\n\t\treturn dummy.next;\n\t}\n};
