class Solution {\npublic:\n\tvoid rotate(vector<vector<int>>& A) {\n\t\tint N = A.size();\n\t\tfor (int i = 0; i < N; ++i)\n\t\t{\n\t\t\tfor (int j = i; j < N; ++j)\n\t\t\t{\n\t\t\t\tstd::swap(A[i][j], A[j][i]);\n\t\t\t}\n\t\t}\n\t\tfor (int i = 0; i < N; ++i)\n\t\t{\n\t\t\tfor (int j = 0; j < N / 2; ++j)\n\t\t\t{\n\t\t\t\tstd::swap(A[i][j], A[i][N - 1 - j]);\n\t\t\t}\n\t\t}\n\t}\n};
