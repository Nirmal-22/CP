class Solution {\npublic:\n\tvector<int> topKFrequent(vector<int>& nums, int k) {\n\t\tunordered_map<int, int> freq;\n\t\tfor (int num : nums) freq[num]++;\n\t\tpriority_queue<\n\t\t\tpair<int, int>,\n\t\t\tvector<pair<int, int>>,\n\t\t\tgreater<>\n\t\t> minHeap;\n\t\tfor (const auto& [num, count] : freq) {\n\t\t\tminHeap.emplace(count, num);\n\t\t\tif (minHeap.size() > k)\n\t\t\t\tminHeap.pop();\n\t\t}\n\t\tvector<int> res;\n\t\twhile (!minHeap.empty()) {\n\t\t\tres.push_back(minHeap.top().second);\n\t\t\tminHeap.pop();\n\t\t}\n\t\treturn res;\n\t}\n};
