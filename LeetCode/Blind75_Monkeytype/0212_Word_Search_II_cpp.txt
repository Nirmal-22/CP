class Solution {\npublic:\n\tvector<string> findWords(vector<vector<char>>& b, vector<string>& ws) {\n\t\tstruct TrieNode{\n\t\t\tarray<TrieNode*, 26> nxt{};\n\t\t\tstring word{};\n\t\t\t// ~TrieNode(){ for(auto p:nxt) if(p) delete p;}\n\t\t};\n\t\tTrieNode* root= new TrieNode();\n\t\tfor(auto& w: ws){\n\t\t\tTrieNode* cur = root;\n\t\t\tfor(char c:w) {\n\t\t\t\tint i= c - 'a';\n\t\t\t\tif(!cur->nxt[i]) cur->nxt[i] = new TrieNode();\n\t\t\t\tcur = cur->nxt[i];\n\t\t\t}\n\t\t\tcur->word = w;\n\t\t}\n\t\tint m = b.size(), n= b[0].size();\n\t\tvector<string> ans;\n\t\tvector<vector<char>> vis(m, vector<char>(n,0));\n\t\tfunction<void(int,int, TrieNode*)> dfs = [&](int x,int y, TrieNode* node){\n\t\t\tchar ch = b[x][y];\n\t\t\tint idx = ch - 'a';\n\t\t\tnode = node->nxt[idx];\n\t\t\tif(!node) return;\n\t\t\tif(!node->word.empty()) {\n\t\t\t\tans.push_back(node->word);\n\t\t\t\tnode->word.clear();\n\t\t\t}\n\t\t\tvis[x][y] = 1;\n\t\t\tconst int dx[4] = {1,-1,0,0}, dy[4]={0,0,1,-1};\n\t\t\tfor(int k=0;k<4;k++){\n\t\t\t\tint nx=x+dx[k], ny=y+dy[k];\n\t\t\t\tif(nx>=0 && nx<m && ny>=0 && ny<n && !vis[nx][ny])\n\t\t\t\t\tdfs(nx,ny,node);\n\t\t\t}\n\t\t\tvis[x][y] = 0;\n\t\t};\n\t\tfor(int i=0;i<m;i++)\n\t\t\tfor(int j=0;j<n;j++)\n\t\t\t\tdfs(i,j,root);\n\t\t// delete root;\n\t\treturn ans;\n\t}\n};
