class Solution {\npublic:\n\tint maxProduct(vector<int>& nums) {\n\t\tint n = nums.size();\n\t\tif (n == 0)\n\t\t{\n\t\t\treturn 0;\n\t\t}\n\t\tint maxProd = nums[0];\n\t\tint minProd = nums[0];\n\t\tint result = maxProd;\n\t\tfor (int i = 1; i < n; i++) {\n\t\t\tif (nums[i] < 0) {\n\t\t\t\tswap(maxProd, minProd);\n\t\t\t}\n\t\t\tmaxProd = max(nums[i], maxProd * nums[i]);\n\t\t\tminProd = min(nums[i], minProd * nums[i]);\n\t\t\tresult = max(result, maxProd);\n\t\t}\n\t\treturn result;\n\t}\n};
