class Solution {\npublic:\n\tvector<vector<int>> merge(vector<vector<int>>& it) {\n\t\tif(it.empty()) return {};\n\t\tsort(it.begin(), it.end());\n\t\tvector<vector<int>> merged;\n\t\tmerged.push_back(it[0]);\n\t\tfor(int i=1;i<it.size();i++){\n\t\t\tauto &last = merged.back();\n\t\t\tif(it[i][0] <= last[1])\n\t\t\t\tlast[1] = max(last[1], it[i][1]);\n\t\t\telse\n\t\t\t\tmerged.push_back(it[i]);\n\t\t}\n\t\treturn merged;\n\t}\n};
