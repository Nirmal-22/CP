class Solution {\npublic:\n\tint eraseOverlapIntervals(vector<vector<int>>& it) {\n\t\tif(it.empty()) return 0;\n\t\tsort(it.begin(), it.end(), [](auto &a, auto &b){ return a[1] < b[1]; });\n\t\tint removed = 0;\n\t\tint lastEnd = it[0][1];\n\t\tfor(int i=1; i<it.size(); i++){\n\t\t\tif(it[i][0] < lastEnd) removed++;\n\t\t\telse lastEnd = it[i][1];\n\t\t}\n\t\treturn removed;\n\t}\n};
