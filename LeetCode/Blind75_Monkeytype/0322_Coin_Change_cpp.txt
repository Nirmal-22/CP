class Solution {\npublic:\n\tint coinChange(vector<int>& c, int s) {\n\t\tvector<int> dp(s+1, s+1);\n\t\tdp[0] = 0;\n\t\tfor(int i=1;i<=s; i++)\n\t\t{\n\t\t\tfor(int coin: c){\n\t\t\t\tif(i >= coin){\n\t\t\t\t\tdp[i] = min(dp[i], dp[i-coin]+1);\n\t\t\t\t}\n\t\t\t}\n\t\t}\n\t\treturn dp[s] > s ? -1:dp[s];\n\t}\n};
