class Solution {\npublic:\n\tint countSubstrings(string s) {\n\t\tint n = s.size(), ans = 0;\n\t\t//2n-1 centers (n single chars + n-1 gaps)\n\t\tfor (int center = 0; center < 2*n - 1; ++center) {\n\t\t\tint l = center / 2;\n\t\t\tint r = l + center % 2;\n\t\t\twhile (l >= 0 && r < n && s[l] == s[r]) {\n\t\t\t\t++ans;\n\t\t\t\t--l; ++r;\n\t\t\t}\n\t\t}\n\t\treturn ans;\n\t}\n};
