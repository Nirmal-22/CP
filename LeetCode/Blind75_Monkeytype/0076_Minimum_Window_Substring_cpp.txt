class Solution {\npublic:\n\tstring minWindow(string s, string t) {\n\t\tif(t.empty() || s.empty()) return "";\n\t\tvector<int> f(128,0);\n\t\tfor(auto c:t) f[c]++;\n\t\tint req=t.size(), l=0, min=INT_MAX, start=0;\n\t\tfor(int r=0;r<(int)s.size(); r++)\n\t\t{\n\t\t\tif(f[s[r]]-- > 0) --req;\n\t\t\twhile(req == 0)\n\t\t\t{\n\t\t\t\tif(r-l+1 < min)\n\t\t\t\t{\n\t\t\t\t\tmin = r-l+1;\n\t\t\t\t\tstart = l;\n\t\t\t\t}\n\t\t\t\tif(++f[s[l]] > 0) ++req;\n\t\t\t\tl++;\n\t\t\t}\n\t\t}\n\t\treturn min == INT_MAX ? "" : s.substr(start, min);\n\t}\n};
