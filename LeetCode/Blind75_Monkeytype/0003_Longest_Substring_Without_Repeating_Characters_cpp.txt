class Solution {\npublic:\n\tint lengthOfLongestSubstring(string s) {\n\t\tvector<int> last(256,-1);\n\t\tint start=0, maxlen=0;\n\t\tfor(int i=0;i<s.size();i++){\n\t\t\tstart = max(start, last[s[i]]+1);\n\t\t\tmaxlen = max(maxlen, i-start+1);\n\t\t\tlast[s[i]] = i;\n\t\t}\n\t\treturn maxlen;\n\t}\n};
