class Solution {\npublic:\n\tint uniquePaths(int m, int n) {\n\t\tvector<vector<int>> dp(m, vector<int>(n,0));\n\t\tfor(int i=0;i<m;i++) dp[i][0]=1;\n\t\tfor(int j=0;j<n;j++) dp[0][j]=1;\n\t\tfor(int i=1;i<m;i++){\n\t\t\tfor(int j=1;j<n; j++){\n\t\t\t\tdp[i][j] = dp[i-1][j] + dp[i][j-1];\n\t\t\t}\n\t\t}\n\t\treturn dp[m-1][n-1];\n\t}\n};
